def qkv_rope_fused(
    X, Wqkv, Bqkv, CosSin, Out,
    S: tl.constexpr, D: tl.constexpr,
    BM: tl.constexpr = 128,
    BN: tl.constexpr = 128,
    BK: tl.constexpr = 64,
    STAGES: tl.constexpr = 4):

  pid = tl.program_id(0)
  acc = tl.zeros((BM, BN), tl.float32)

  for k in tl.range(0, D, BK,
                    num_stages=STAGES):
    a = tl.load(X + ...)
    b = tl.load(Wqkv + ...)
    acc = tl.dot(a, b, acc)

  # epilogue: bias + rope, no round trip
  acc += tl.load(Bqkv + ...)
  cs = tl.load(CosSin + ...)
  acc = rope_inplace(acc, cs)
  tl.store(Out + ..., acc.to(tl.float16))
